    <script>
        const surveySteps = [
            {
                key: 'cycle',
                stepTitle: '1 / 4 단계: 주기 변화',
                question: '안녕하세요! 여성 건강 AI 문진 도우미입니다. 먼저, 최근 생리 주기에 변화가 있었나요?',
                quickReplies: ['규칙적인 편이에요 (21~35일 주기)', '예정일보다 1주 이상 늦어졌어요', '생리 기간이 아닌데 부정출혈이 있어요', '3개월 이상 생리가 없어요 (무월경)']
            },
            {
                key: 'pain',
                stepTitle: '2 / 4 단계: 생리통 강도',
                question: '다음으로 생리통의 강도는 어느 정도인가요? (진통제 복용 여부나 일상생활 지장 정도를 알려주세요.)',
                quickReplies: ['1~3점: 통증이 거의 없거나 참을 만해요', '4~6점: 진통제를 먹으면 일상생활이 가능해요', '7~10점: 진통제가 잘 안 듣고 일상생활이 어려워요']
            },
            {
                key: 'bleeding',
                stepTitle: '3 / 4 단계: 출혈량',
                question: '생리 중 출혈량은 평소와 비교했을 때 어떤가요?',
                quickReplies: ['보통이에요 (3~4시간마다 패드 교체)', '평소보다 조금 많은 편이에요', '1~2시간마다 대형 패드를 적실 정도로 많아요', '엄지손톱보다 큰 덩어리혈이 자주 나와요']
            },
            {
                key: 'other',
                stepTitle: '4 / 4 단계: 동반 증상',
                question: '마지막 질문입니다. 생리 전후로 함께 나타나는 동반 증상이 있나요?',
                quickReplies: ['특별한 동반 증상은 없어요', '어지러움, 숨참 등 빈혈 기운이 있어요', '38도 이상의 발열이나 오한이 있어요', '심한 메스꺼움, 구토 또는 골반 통증이 있어요']
            }
        ];

        let currentStep = 0;
        let answers = { cycle: '', pain: '', bleeding: '', other: '' };
        let summaryTextForCopy = '';

        const chatMessages = document.getElementById('chat-messages');
        const quickReplies = document.getElementById('quick-replies');
        const userInput = document.getElementById('user-input');

        function initChat() {
            currentStep = 0;
            answers = { cycle: '', pain: '', bleeding: '', other: '' };
            chatMessages.innerHTML = '';
            ['cycle', 'pain', 'bleeding', 'other'].forEach(k => document.getElementById(`summary-${k}`).textContent = '대기 중...');
            document.getElementById('report-placeholder').classList.remove('hidden');
            document.getElementById('report-content').classList.add('hidden');
            document.getElementById('quick-replies-container').classList.remove('hidden');
            updateProgress(1);
            askQuestion(0);
        }

        function updateProgress(step) {
            for (let i = 1; i <= 5; i++) {
                document.getElementById(`prog-${i}`).classList.toggle('active', i <= step);
                document.getElementById(`label-${i}`).classList.toggle('active', i <= step);
            }
            document.getElementById('step-indicator-text').textContent = step <= 4 ? surveySteps[step - 1].stepTitle : '문진 완료 • 결과 리포트 생성됨';
        }

        function askQuestion(idx) {
            const step = surveySteps[idx];
            appendMsg('bot', step.question);
            quickReplies.innerHTML = '';
            step.quickReplies.forEach(opt => {
                const btn = document.createElement('button');
                btn.type = 'button';
                btn.className = 'chip-btn';
                btn.textContent = opt;
                btn.onclick = () => handleAnswer(opt);
                quickReplies.appendChild(btn);
            });
        }

        function appendMsg(sender, text) {
            const div = document.createElement('div');
            div.className = `msg-row ${sender}`;
            div.innerHTML = `<div class="msg-bubble">${text}</div>`;
            chatMessages.appendChild(div);
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        function handleAnswer(text) {
            appendMsg('user', text);
            if (currentStep < surveySteps.length) {
                const key = surveySteps[currentStep].key;
                answers[key] = text;
                document.getElementById(`summary-${key}`).textContent = text;
                currentStep++;
                if (currentStep < surveySteps.length) {
                    updateProgress(currentStep + 1);
                    askQuestion(currentStep);
                } else {
                    updateProgress(5);
                    document.getElementById('quick-replies-container').classList.add('hidden');
                    showFinalResult();
                }
            }
        }

        function showFinalResult() {
            const combined = `${answers.cycle} ${answers.pain} ${answers.bleeding} ${answers.other}`;
            let level = 'GREEN', title = '🟢 정상 범위 / 생활 습관 관리', desc = '특이 위험 신호가 보이지 않습니다. 충분한 휴식과 온찜질로 컨디션을 관리해 주세요.';
            
            if (combined.includes('7~10') || combined.includes('1~2시간') || combined.includes('덩어리혈') || combined.includes('부정출혈') || combined.includes('3개월') || combined.includes('어지러움') || combined.includes('발열')) {
                level = 'RED';
                title = '🔴 즉시 내원 필요 (산부인과 진료 권고)';
                desc = '월경과다, 심한 통증, 부정출혈 또는 전신 동반 증상이 감지되었습니다. 빠른 시일 내에 산부인과 전문의 진료를 받으시길 권장합니다.';
            } else if (combined.includes('4~6') || combined.includes('1주 이상') || combined.includes('조금 많은')) {
                level = 'YELLOW';
                title = '🟡 경과 관찰 및 진료 권장';
                desc = '일시적 변화일 수 있으나 증상이 다음 주기에도 반복되면 산부인과 상담을 받아보세요.';
            }

            const recBox = document.getElementById('recommendation-box');
            recBox.className = `rec-box ${level.toLowerCase()}`;
            document.getElementById('rec-title').textContent = title;
            document.getElementById('rec-desc').textContent = desc;

            summaryTextForCopy = `[생리 증상 문진 요약]\n• 권고 등급: ${title}\n• 주기 양상: ${answers.cycle}\n• 통증 강도: ${answers.pain}\n• 출혈 양상: ${answers.bleeding}\n• 동반 증상: ${answers.other}`;
            document.getElementById('doctor-summary-text').textContent = summaryTextForCopy;

            document.getElementById('report-placeholder').classList.add('hidden');
            document.getElementById('report-content').classList.remove('hidden');
            document.getElementById('live-status-badge').textContent = '분석 완료';
            appendMsg('bot', `문진이 완료되었습니다! 우측 결과 리포트에서 [${title}] 결과와 의사 전달용 요약본을 확인해 주세요.`);
        }

        function copySummaryText() {
            navigator.clipboard.writeText(summaryTextForCopy).then(() => alert('요약본이 복사되었습니다!'));
        }

        document.getElementById('chat-form').addEventListener('submit', e => {
            e.preventDefault();
            if (userInput.value.trim()) {
                handleAnswer(userInput.value.trim());
                userInput.value = '';
            }
        });
        document.getElementById('reset-btn').addEventListener('click', initChat);
        window.addEventListener('DOMContentLoaded', initChat);
    </script>
